# All commands use: uint32 total packet length, uint16 command type, then parameters.
COMMAND PLUTO_SDR CONFIG BIG_ENDIAN "Configure Pluto receive settings"
  APPEND_PARAMETER LENGTH 32 UINT 36 36 36 "Total packet bytes"
  APPEND_ID_PARAMETER TYPE 16 UINT 100 100 100 "Configure"
  APPEND_PARAMETER CENTER_HZ 64 UINT 70000000 6000000000 915000000 "Center Hz"
  APPEND_PARAMETER SPAN_HZ 32 UINT 50000 20000000 1600000 "Displayed span Hz"
  APPEND_PARAMETER SAMPLE_RATE 32 UINT 520833 20000000 2000000 "Samples per second"
  APPEND_PARAMETER FFT_SIZE 32 UINT 256 8192 2048 "FFT length"
  APPEND_PARAMETER GAIN_DB 32 FLOAT 0 70 40 "Manual gain dB"
  APPEND_PARAMETER AVERAGES 32 FLOAT 1 32 8 "FFT windows per update"
  APPEND_PARAMETER AGC 8 UINT 0 1 0 "AGC 0 manual, 1 automatic"
  APPEND_PARAMETER FPS 8 UINT 1 5 2 "Display frames per second"

# V2 CONFIG separates digital display span and analog RF receive filter width.
# Legacy CONFIG (ID 100) remains accepted by the bridge.
COMMAND PLUTO_SDR CONFIG_V2 BIG_ENDIAN "Set Pluto display and hardware settings"
  APPEND_PARAMETER LENGTH 32 UINT 40 40 40 "Total packet bytes"
  APPEND_ID_PARAMETER TYPE 16 UINT 104 104 104 "Configure V2"
  APPEND_PARAMETER CENTER_HZ 64 UINT 70000000 6000000000 915000000 "Center Hz"
  APPEND_PARAMETER SPAN_HZ 32 UINT 50000 20000000 1600000 "Displayed span Hz"
  APPEND_PARAMETER SAMPLE_RATE 32 UINT 520833 20000000 2000000 "Samples per second"
  APPEND_PARAMETER FFT_SIZE 32 UINT 256 8192 2048 "FFT length"
  APPEND_PARAMETER GAIN_DB 32 FLOAT 0 70 40 "Manual gain dB"
  APPEND_PARAMETER AVERAGES 32 FLOAT 1 32 8 "FFT windows per update"
  APPEND_PARAMETER AGC 8 UINT 0 1 0 "AGC 0 manual, 1 automatic"
  APPEND_PARAMETER FPS 8 UINT 1 5 2 "Display FFT frames/second"
  APPEND_PARAMETER RF_BANDWIDTH_HZ 32 UINT 200000 20000000 1600000 "Hardware RF filter bandwidth Hz"
